func intsAny(v []int)[]any {o:=make([]any,len(v));for i,n:=range v{o[i]=n};return o}
func partyFresh(t *testing.T,release,charge,right int,trace bool,mutation string) map[string]any {
 g:=newTestGame(t);g.Configure(settings.Legacy());g.ResearchDemoEnabled=true;g.ResearchPartyMutation=mutation
 g.Display.Content.Texts["S_GAMEOVER2"]=append([]byte(nil),postConfig.Jingle...)
 researchScrollTexts(g)
 rows:=[]map[string]any{};boundaries:=[]map[string]any{};events:=[]map[string]any{};drain:=0;scored:=false;prefixScoreFalse:=true;prefixResetOne:=true;roles:=map[string]int{};physicsReads:=map[string]bool{}
 g.Physics.ResearchTrace=func(role string,v ...int){if trace&&drain==0 {raw,_:=json.Marshal(append([]any{role},intsAny(v)...));physicsReads[string(raw)]=true}}
 snap:=func()map[string]any{v:=postSnapshot(g);v["VISAKEYS"]=false;return v}
 g.ResearchObserve=func(role string){
  if drain==0 {roles[role]++}
  if role=="drain_entry" {if drain==0 {drain=int(g.Tick);scored=g.ScoreChanged}}
  if role=="wait_before:PARTY_ON_TASK1"&&g.Tick==35998&&mutation=="clear_before_body" {g.partyFlash=false}
  if role=="party_guard"&&mutation=="clear_party" {g.partyFlash=false}
  if trace&&(int(g.Tick)>=drain&&drain>0||role=="drain_entry") {v:=snap();v["boundary"]=role;boundaries=append(boundaries,v)}
 }
 for n:=1;n<=37200&&!g.ResearchQuit;n++ {
  in:=physics.Inputs{Down:n<release&&(charge==0||n>=release-charge),Release:n==release,Right:n>=release&&n<release+right}
  if err:=g.Sync(in);err!=nil {t.Fatal(err)}
  if drain==0||n==drain {prefixScoreFalse=prefixScoreFalse&&!g.ScoreChanged;prefixResetOne=prefixResetOne&&g.ResearchResetCount==1}
  if trace&&(n==1||n==release-1||n==release||n>=drain&&drain>0) {v:=snap();v["input"]=in;rows=append(rows,v)}
  if trace&&len(g.Events)>0&&n>=drain&&drain>0 {events=append(events,map[string]any{"calculation":n,"events":append([]Event(nil),g.Events...)})}
  if !trace&&drain>0 {break}
  if mutation!=""&&n>=36000 {break}
 }
 return map[string]any{"prefix_score_false":prefixScoreFalse,"prefix_reset_one":prefixResetOne,"prefix_roles":roles,"prefix_physics_reads":physicsReads,"release":release,"charge":charge,"right_duration":right,"drain":drain,"scored":scored,"rows":rows,"boundaries":boundaries,"event_rows":events,"final":snap(),"mutation":mutation}
}

 for _,charge:=range []int{0,22,18,32} {for release:=35580;release<=35880;release++ {a:=partyFresh(t,release,charge,0,false,"");attempts=append(attempts,a);if a["drain"]==35968&&!a["scored"].(bool) {found=a;break}};if found!=nil{break}}

 a,b:=partyFresh(t,release,charge,0,true,""),partyFresh(t,release,charge,0,true,"")
 if !reflect.DeepEqual(a,b){t.Fatal("nondeterministic target")}
 out:=map[string]any{"attempts":attempts,"witness":a,"deterministic_equal":true,"neighbors":[]map[string]any{partyFresh(t,35838,charge,0,true,""),partyFresh(t,35841,charge,0,true,"")},"release_plus_one":partyFresh(t,release+1,charge,0,true,""),"clear_party":partyFresh(t,release,charge,0,true,"clear_party"),"delay_party":partyFresh(t,release,charge,0,true,"delay_party"),"clear_before_body":partyFresh(t,release,charge,0,true,"clear_before_body")}
 raw,_:=json.Marshal(out);if err:=os.WriteFile(os.Getenv("PF_PARTY_OUTPUT"),raw,0600);err!=nil{t.Fatal(err)}
 t.Log("target",release,charge,a["drain"],"final",a["final"].(map[string]any)["calculation"])
}
