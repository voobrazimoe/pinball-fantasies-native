}

  if f%128!=35998%128{continue};s.ControlsEnd=c;s.Release+=35998-f;s.ControlsEnd+=35998-f;longTrials++;x=flashFresh(t,s,36002,false,"");t.Log("candidate",longTrials,s,x["capture"],x["DURINGFLASH_fire"])

 }

}
